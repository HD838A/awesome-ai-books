10.2.2 子 Agent 系统提示



安全审查 Agent：

# 安全审查 Agent 系统提示



你是一个专业的安全工程师，负责审查代码中的安全漏洞。




审查重点





SQL 注入



  • 检查所有 SQL 查询是否使用参数化查询
  • 检查 ORM 使用是否正确
  • 检查用户输入是否经过验证





XSS 攻击



  • 检查输出是否经过转义
  • 检查 Content-Type 设置
  • 检查 CSP 策略





认证和授权



  • 检查密码存储是否使用哈希
  • 检查 Session 管理
  • 检查权限检查逻辑





工具使用




使用以下工具获取代码：


  • web_search: 搜索已知漏洞模式
  • terminal: 运行安全扫描工具





输出




对于每个发现的问题，提供：


  • 文件和行号
  • 问题描述
  • 严重程度（Critical/High/Medium/Low）
  • 修复建议
  • 参考资料（CVE 或 CWE）

@dataclass

@dataclass

class ReviewResult:

    category: str

def fetch_code(code_or_url: str) -> Dict[str, str]:

    """

    获取代码内容，支持直接代码或 URL

    """

    if code_or_url.startswith("http"):

        # 从 URL 获取代码

        import httpx

        response = httpx.get(code_or_url)

        return {"content": response.text, "source": "url"}

    else:

        # 直接代码内容

        return {"content": code_or_url, "source": "direct"}



def code_review_system(code_or_url: str, language: str = "python") -> str:

    主审查流程：

    # Step 1: 获取代码

    import json

    results = []

    for task in tasks:

            goal=task["goal"],

            context=task["context"],

    return report



def generate_report(results: List[Dict]) -> str:

    """汇总子 Agent 结果，生成统一报告"""

    

    report = "# 代码审查报告\n\n"

    

    # 执行摘要

    all_issues = []

    for result in results:

        for issue in result.get("issues", []):

            all_issues.append(issue)

    

    total_issues = len(all_issues)

    report += "## 执行摘要\n\n"

    report += f"- 总问题数：{total_issues}\n"

    report += f"- 审查范围：{len(results)} 个领域\n\n"

    

    # 分类统计

    by_severity = {"Critical": [], "High": [], "Medium": [], "Low": []}

    for issue in all_issues:

        severity = issue.get("severity", "Low")

        if severity in by_severity:

            by_severity[severity].append(issue)

    

    # 详细问题

    for severity in ["Critical", "High", "Medium", "Low"]:

        issues = by_severity[severity]

        if issues:

            emoji = {"Critical": "🔴", "High": "🟠", "Medium": "🟡", "Low": "🟢"}[severity]

            report += f"## {emoji} {severity} 问题\n\n"

            report += "| # | 位置 | 问题 | 建议 |\n"

            report += "|---|------|------|------|\n"

            for i, issue in enumerate(issues, 1):

                report += f"| {i} | {issue.get('location', 'N/A')} | {issue.get('description', '')} | {issue.get('fix', '')} |\n"

            report += "\n"

    

    # 添加各领域总结

    report += "## 审查总结\n\n"

    for i, result in enumerate(results):

        summary = result.get("summary", "无总结")

        report += f"- 领域 {i+1}：{summary}\n"

    

    return report
