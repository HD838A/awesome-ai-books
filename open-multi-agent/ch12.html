第 12 章：扩展开发

让框架适配你的世界

框架提供了开箱即用的功能，但真实世界总是有各种定制需求。

本章我们将学习如何扩展框架：

  • 自定义工具：接入任何外部系统
  • 自定义 Adapter：支持任何 LLM
  • 自定义 MemoryStore：持久化到任何存储
  • 自定义 Scheduler：实现自己的调度策略

12.2 自定义工具

12.2.1 defineTool 详解

框架使用 defineTool 工厂函数创建工具：

import { defineTool, z } from '@jackchen_me/open-multi-agent'

export const weatherTool = defineTool({
  name: 'get_weather',
  description: '获取指定城市的当前天气',
  inputSchema: z.object({
    city: z.string().describe('城市名称'),
    unit: z.enum(['celsius', 'fahrenheit']).default('celsius'),
  }),
  execute: async (input, context) => {
    // input 是 Zod 验证后的类型安全对象
    const { city, unit } = input

    // 调用外部 API
    const response = await fetch(`https://api.weather.example.com?city=${city}`)
    const data = await response.json()

    return {
      data: `当前 ${city} 天气：${data.temp}${unit === 'celsius' ? '°C' : '°F'}`,
      isError: false,
    }
  },
})

12.2.4 完整示例：GitHub 工具

export const githubTool = defineTool({
  name: 'github',
  description: '与 GitHub API 交互：搜索仓库、获取 issue、创建 PR 等',

  inputSchema: z.object({
    action: z.enum(['search_repos', 'get_issue', 'create_issue', 'create_pr']),
    params: z.record(z.unknown()),
  }),

  execute: async ({ action, params }, { signal }) => {
    const token = process.env.GITHUB_TOKEN
    if (!token) {
      return { data: 'GITHUB_TOKEN 未设置', isError: true }
    }

    const endpoints = {
      search_repos: `/search/repositories?q=${params.q}`,
      get_issue: `/repos/${params.owner}/${params.repo}/issues/${params.number}`,
      create_issue: `/repos/${params.owner}/${params.repo}/issues`,
      create_pr: `/repos/${params.owner}/${params.repo}/pulls`,
    }

    const response = await fetch(`https://api.github.com${endpoints[action]}`, {
      method: action.startsWith('create_') ? 'POST' : 'GET',
      headers: {
        Authorization: `Bearer ${token}`,
        Accept: 'application/vnd.github+json',
      },
      body: action.startsWith('create_') ? JSON.stringify(params.body) : undefined,
      signal,
    })

    const data = await response.json()

    if (!response.ok) {
      return { data: `GitHub API 错误: ${data.message}`, isError: true }
    }

    return { data: JSON.stringify(data), isError: false }
  },
})

12.3.3 实现新的 Adapter

import type { LLMAdapter, LLMMessage, LLMResponse, ChatOptions } from '@jackchen_me/open-multi-agent'

class CustomModelAdapter implements LLMAdapter {
  constructor(
    private apiKey: string,
    private baseURL: string = 'https://api.example.com',
  ) {}

  async chat(
    messages: LLMMessage[],
    options?: Partial<ChatOptions>,
    signal?: AbortSignal,
  ): Promise<LLMResponse> {
    const response = await fetch(`${this.baseURL}/chat`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        Authorization: `Bearer ${this.apiKey}`,
      },
      body: JSON.stringify({
        model: options?.model ?? 'default',
        messages: messages.map(m => ({
          role: m.role,
          content: m.content,
        })),
        temperature: options?.temperature ?? 0.7,
        max_tokens: options?.maxTokens ?? 4096,
      }),
      signal,
    })

    if (!response.ok) {
      throw new Error(`API 错误: ${response.status}`)
    }

    const data = await response.json()

    return {
      content: data.choices[0].message.content,
      stopReason: data.choices[0].finish_reason,
      tokenUsage: {
        input_tokens: data.usage.prompt_tokens,
        output_tokens: data.usage.completion_tokens,
      },
    }
  }
}

12.3.4 注册自定义 Adapter

// 修改 createAdapter 工厂
import { createAdapter } from '@jackchen_me/open-multi-agent'

export async function createCustomAdapter(
  provider: string,
  apiKey?: string,
  baseURL?: string,
): Promise<LLMAdapter> {
  switch (provider) {
    case 'anthropic':
    case 'openai':
    case 'gemini':
      return createAdapter(provider as SupportedProvider, apiKey, baseURL)

    case 'custom':
      return new CustomModelAdapter(apiKey!, baseURL)

    default:
      throw new Error(`不支持的 provider: ${provider}`)
  }
}

12.4 自定义 MemoryStore

12.4.1 MemoryStore 接口

interface MemoryStore {
  get(key: string): Promise<MemoryEntry | null>
  set(key: string, value: string, metadata?: Record<string, unknown>): Promise<void>
  list(): Promise<MemoryEntry[]>
}

interface MemoryEntry {
  key: string
  value: string
  metadata: Record<string, unknown>
  createdAt: Date
  updatedAt: Date
}

12.4.2 Redis 实现

import { createClient, type RedisClientType } from 'redis'
import type { MemoryStore, MemoryEntry } from '@jackchen_me/open-multi-agent'

class RedisMemoryStore implements MemoryStore {
  private client: RedisClientType

  constructor(url: string = process.env.REDIS_URL ?? 'redis://localhost:6379') {
    this.client = createClient({ url })
  }

  async connect() {
    await this.client.connect()
  }

  async get(key: string): Promise<MemoryEntry | null> {
    const data = await this.client.hGetAll(key)
    if (!data.value) return null

    return {
      key,
      value: data.value,
      metadata: JSON.parse(data.metadata ?? '{}'),
      createdAt: new Date(data.createdAt),
      updatedAt: new Date(data.updatedAt),
    }
  }

  async set(
    key: string,
    value: string,
    metadata?: Record<string, unknown>,
  ): Promise<void> {
    const now = new Date().toISOString()
    const existing = await this.client.hGet(key, 'createdAt')

    await this.client.hSet(key, {
      value,
      metadata: JSON.stringify(metadata ?? {}),
      createdAt: existing ?? now,
      updatedAt: now,
    })
  }

  async list(): Promise<MemoryEntry[]> {
    const keys = await this.client.keys('*')
    return Promise.all(keys.map(k => this.get(k)))
  }

  async disconnect() {
    await this.client.disconnect()
  }
}

12.4.3 PostgreSQL 实现

import pg from 'pg'
import type { MemoryStore, MemoryEntry } from '@jackchen_me/open-multi-agent'

class PostgresMemoryStore implements MemoryStore {
  private pool: pg.Pool

  constructor(connectionString: string) {
    this.pool = new pg.Pool({ connectionString })
  }

  async initialize() {
    await this.pool.query(`
      CREATE TABLE IF NOT EXISTS memory (
        key TEXT PRIMARY KEY,
        value TEXT NOT NULL,
        metadata JSONB DEFAULT '{}',
        created_at TIMESTAMP DEFAULT NOW(),
        updated_at TIMESTAMP DEFAULT NOW()
      )
    `)
  }

  async get(key: string): Promise<MemoryEntry | null> {
    const result = await this.pool.query(
      'SELECT * FROM memory WHERE key = $1',
      [key]
    )
    if (result.rows.length === 0) return null

    const row = result.rows[0]
    return {
      key: row.key,
      value: row.value,
      metadata: row.metadata,
      createdAt: row.created_at,
      updatedAt: row.updated_at,
    }
  }

  async set(key: string, value: string, metadata?: Record<string, unknown>): Promise<void> {
    await this.pool.query(`
      INSERT INTO memory (key, value, metadata, updated_at)
      VALUES ($1, $2, $3, NOW())
      ON CONFLICT (key) DO UPDATE SET
        value = EXCLUDED.value,
        metadata = EXCLUDED.metadata,
        updated_at = NOW()
    `, [key, value, JSON.stringify(metadata ?? {})])
  }

  async list(): Promise<MemoryEntry[]> {
    const result = await this.pool.query('SELECT * FROM memory ORDER BY updated_at DESC')
    return result.rows.map(row => ({
      key: row.key,
      value: row.value,
      metadata: row.metadata,
      createdAt: row.created_at,
      updatedAt: row.updated_at,
    }))
  }
}

12.5 自定义 Scheduler

12.5.1 SchedulingStrategy 接口

框架内置四种调度策略：

type SchedulingStrategy =
  | 'round-robin'       // 轮询分配
  | 'least-busy'        // 最空闲优先
  | 'capability-match'  // 能力匹配
  | 'dependency-first'  // 依赖优先（默认）

12.5.2 实现自定义策略

import type { Scheduler, Agent, Task } from '@jackchen_me/open-multi-agent'

class PriorityScheduler implements Scheduler {
  selectAgent(availableAgents: Agent[], task: Task, pendingTasks: Task[]): Agent | null {
    // 优先级调度：高优先级任务优先分配给最快的 Agent
    const priority = task.metadata?.priority ?? 0

    if (priority >= 10) {
      // 高优先级：选择负载最低的 Agent
      return availableAgents.sort((a, b) =>
        a.getLoad() - b.getLoad()
      )[0]
    }

    // 普通优先级：按负载均匀分配
    return availableAgents[Math.floor(Math.random() * availableAgents.length)]
  }
}

12.5.3 注册自定义 Scheduler

const orchestrator = new OpenMultiAgent({
  defaultModel: 'claude-sonnet-4-6',
  scheduler: new PriorityScheduler(),  // 注入自定义调度器
})

12.6 中间件与拦截器

12.6.1 工具执行拦截器

class LoggingToolExecutor extends ToolExecutor {
  async executeBatch(calls, context) {
    console.log(`[ToolExecutor] 即将执行 ${calls.length} 个工具调用`)

    const startTime = Date.now()
    const results = await super.executeBatch(calls, context)
    const duration = Date.now() - startTime

    console.log(`[ToolExecutor] 完成，耗时 ${duration}ms`)

    return results
  }
}

12.6.2 LLM 调用拦截器

class MetricsAdapter implements LLMAdapter {
  constructor(private adapter: LLMAdapter) {}

  async chat(messages, options, signal) {
    const startTime = Date.now()

    try {
      const result = await this.adapter.chat(messages, options, signal)

      metrics.record('llm.success', {
        model: options?.model,
        duration: Date.now() - startTime,
      })

      return result
    } catch (error) {
      metrics.record('llm.error', {
        model: options?.model,
        error: error.message,
      })
      throw error
    }
  }
}

12.9 最佳实践

12.9.1 类型安全

始终使用 TypeScript 和 Zod：

// ✅ 正确：完整的类型安全
const tool = defineTool({
  name: 'search',
  inputSchema: z.object({
    query: z.string().min(1),
    limit: z.number().int().min(1).max(100).default(10),
  }),
  execute: async (input) => {
    // input.query 是 string，input.limit 是 number
    // TypeScript 知道你在这里的类型
  },
})

// ❌ 错误：放弃类型安全
const tool = defineTool({
  name: 'search',
  inputSchema: z.any(),
  execute: async (input: any) => { ... },
})

12.9.4 性能考虑

execute: async (input, context) => {
  // 支持 AbortSignal
  const result = await fetchSomething(input, { signal: context.signal })

  // 添加超时
  const withTimeout = Promise.race([
    fetchSomething(input),
    new Promise((_, reject) =>
      setTimeout(() => reject(new Error('Timeout')), 30000)
    ),
  ])

  return await withTimeout
}

本章小结

  1. 自定义工具：defineTool() + Zod schema + execute 函数，最常用的扩展方式。
  1. 自定义 Adapter：实现 LLMAdapter 接口或继承现有 Adapter，添加新模型支持。
  1. 自定义 MemoryStore：实现 get/set/list 接口，持久化到 Redis、PostgreSQL 等。
  1. 自定义 Scheduler：实现 selectAgent 方法，定制任务分配策略。
  1. 拦截器模式：包装 ToolExecutor 或 LLMAdapter，添加日志、指标等功能。
  1. 发布扩展：标准化包结构，通过 peerDependencies 依赖主框架。
