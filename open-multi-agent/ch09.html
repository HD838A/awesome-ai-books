第 9 章：高级特性

让系统更健壮

前面的章节介绍了 Open Multi-Agent 的核心功能。本章我们将探索一些高级特性，让你的 Agent 系统更加：

  • 可观测：了解系统内部发生了什么
  • 可控：随时中断长时间运行的任务
  • 高效：批量并行处理大量任务
  • 可靠：优雅处理各种异常情况

9.1 追踪系统（Tracing）

9.1.1 为什么需要追踪？

当一个复杂任务涉及多个 Agent、多轮对话、多次工具调用时，调试变得非常困难：

用户请求
    ↓
团队协作（3 个 Agent）
    ↓
每个 Agent 进行多轮对话
    ↓
每次对话可能调用 10+ 次工具
    ↓
Token 消耗几何级增长

如果没有追踪，你就像在黑暗中操作。

9.1.2 TraceEvent 结构

interface TraceEvent {
  type: 'llm_call' | 'tool_call' | 'task_start' | 'task_complete'
  runId: string          // 唯一运行 ID
  taskId?: string        // 任务 ID
  agent: string          // Agent 名称
  model: string          // 模型名称
  turn?: number          // 对话轮次
  tokens?: TokenUsage    // Token 消耗
  startMs: number        // 开始时间戳
  endMs: number          // 结束时间戳
  durationMs: number     // 持续时间（毫秒）
}

9.1.3 基本使用

const result = await orchestrator.runTeam(team, goal, {
  onTrace: (event) => {
    console.log(`[${event.type}] ${event.agent} | ${event.durationMs}ms`)

    if (event.type === 'llm_call') {
      console.log(`  模型: ${event.model}`)
      console.log(`  Token: 输入 ${event.tokens?.input_tokens}, 输出 ${event.tokens?.output_tokens}`)
      console.log(`  轮次: ${event.turn}`)
    }
  },
})

9.1.5 集成追踪服务

import { NodeTracerProvider } from '@opentelemetry/sdk-trace-node'
import { ConsoleSpanExporter } from '@opentelemetry/sdk-trace-node'

// 创建追踪提供者
const provider = new NodeTracerProvider()
provider.addSpanProcessor(new ConsoleSpanExporter())
provider.register()

// 追踪回调
const traces: TraceEvent[] = []

result = await orchestrator.runTeam(team, goal, {
  onTrace: (event) => {
    traces.push(event)

    // 发送到追踪服务
    const span = tracer.startSpan(event.type)
    span.setAttributes({
      'agent.name': event.agent,
      'llm.model': event.model,
      'duration.ms': event.durationMs,
    })
    span.end()
  },
})

9.1.6 成本追踪

interface CostSummary {
  totalInputTokens: number
  totalOutputTokens: number
  estimatedCost: number
  byAgent: Map<string, TokenUsage>
  byModel: Map<string, TokenUsage>
}

function calculateCost(result: TeamRunResult): CostSummary {
  const byAgent = new Map<string, TokenUsage>()
  const byModel = new Map<string, TokenUsage>()

  let totalInput = 0
  let totalOutput = 0

  for (const [name, agentResult] of result.agentResults) {
    const usage = agentResult.tokenUsage
    totalInput += usage.input_tokens
    totalOutput += usage.output_tokens

    byAgent.set(name, usage)
    // 按模型汇总...
  }

  // Claude Sonnet 定价：输入 $3/MTok, 输出 $15/MTok
  const estimatedCost =
    (totalInput / 1_000_000) * 3 +
    (totalOutput / 1_000_000) * 15

  return { totalInputTokens: totalInput, totalOutputTokens: totalOutput, estimatedCost, byAgent, byModel }
}

9.2.2 带进度的工具执行

for await (const event of developer.stream('创建 TODO 应用')) {
  switch (event.type) {
    case 'text':
      process.stdout.write(event.data)
      break

    case 'tool_use':
      console.log(`\n🔧 [${new Date().toLocaleTimeString()}] 调用工具: ${event.data.name}`)
      break

    case 'tool_result':
      if (event.data.is_error) {
        console.log(`   ❌ 失败: ${event.data.content}`)
      } else {
        console.log(`   ✅ 成功`)
      }
      break

    case 'loop_detected':
      console.log(`\n⚠️ 警告: 检测到可能的循环`)
      break

    case 'done':
      console.log('\n\n✅ 完成！')
      break
  }
}

9.2.3 带超时的流式处理

import { EventEmitter } from 'events'

const controller = new AbortController()

// 30 秒超时
const timeout = setTimeout(() => {
  console.log('⏰ 超时，中止执行')
  controller.abort()
}, 30000)

try {
  for await (const event of agent.stream('长文本生成', {
    abortSignal: controller.signal,
  })) {
    if (event.type === 'text') {
      process.stdout.write(event.data)
    } else if (event.type === 'done') {
      clearTimeout(timeout)
    }
  }
} catch (error) {
  if (error.name === 'AbortError') {
    console.log('\n⛔ 请求被中止')
  }
}

9.3 批量执行

9.3.1 AgentPool.runParallel

import { AgentPool } from '@jackchen_me/open-multi-agent'

const pool = new AgentPool(3)  // 最多 3 个并发

pool.add(agent1)
pool.add(agent2)
pool.add(agent3)
pool.add(agent4)

// 批量并行执行
const results = await pool.runParallel([
  { agent: 'agent1', prompt: '任务 A' },
  { agent: 'agent2', prompt: '任务 B' },
  { agent: 'agent3', prompt: '任务 C' },
  { agent: 'agent4', prompt: '任务 D' },  // agent4 会等待，因为 maxConcurrency=3
])

// 处理结果
for (const [name, result] of results) {
  console.log(`${name}: ${result.success ? '✅' : '❌'}`)
  if (!result.success) {
    console.log(`  错误: ${result.output}`)
  }
}

9.3.2 批量工具执行

const executor = new ToolExecutor(registry, { maxConcurrency: 5 })

// 一次执行多个工具
const batchResults = await executor.executeBatch([
  { id: 'call-1', name: 'file_read', input: { path: '/tmp/a.txt' } },
  { id: 'call-2', name: 'file_read', input: { path: '/tmp/b.txt' } },
  { id: 'call-3', name: 'bash', input: { command: 'ls /tmp' } },
], context)

// 等待所有结果

  console.log(`${id}: ${result.isError ? '❌' : '✅'}`)
}

9.3.3 批量任务处理

async function processBatch(items: string[]) {
  const results: ProcessedItem[] = []

  const chunks = chunkArray(items, 5)  // 每批 5 个

  for (const chunk of chunks) {
    const batchResults = await pool.runParallel(
      chunk.map((item, i) => ({
        agent: `processor-${i % 3}`,  // 轮换使用 3 个 Agent
        prompt: `处理: ${item}`,
      }))
    )

    for (const [, result] of batchResults) {
      if (result.success) {
        results.push(JSON.parse(result.output))
      }
    }
  }

  return results
}

9.4.3 组合取消信号

// 任一条件触发即取消
const userCancel = new AbortController()
const timeoutCancel = AbortSignal.timeout(5 * 60 * 1000)

const combined = mergeAbortSignals(userCancel.signal, timeoutCancel)

const result = await agent.run(task, {
  abortSignal: combined,
})

// 辅助函数
function mergeAbortSignals(a: AbortSignal, b: AbortSignal): AbortSignal {
  const controller = new AbortController()
  if (a.aborted || b.aborted) { controller.abort(); return controller.signal }

  a.addEventListener('abort', () => controller.abort(), { once: true })
  b.addEventListener('abort', () => controller.abort(), { once: true })

  return controller.signal
}

9.6 完整可观测性示例

9.6.1 构建监控仪表盘

interface Metrics {
  llmCalls: number
  totalTokens: number
  toolCalls: number
  errors: number
  startTime: number
}

async function runWithMonitoring(
  team: Team,
  goal: string,
): Promise<{ result: TeamRunResult; metrics: Metrics }> {
  const metrics: Metrics = {
    llmCalls: 0,
    totalTokens: 0,
    toolCalls: 0,
    errors: 0,
    startTime: Date.now(),
  }

  const result = await orchestrator.runTeam(team, goal, {
    onTrace: (event) => {
      switch (event.type) {
        case 'llm_call':
          metrics.llmCalls++
          metrics.totalTokens +=
            (event.tokens?.input_tokens ?? 0) +
            (event.tokens?.output_tokens ?? 0)
          break
        case 'tool_call':
          metrics.toolCalls++
          break
      }
    },
    onWarning: (msg) => {
      console.warn(`⚠️ 警告: ${msg}`)
    },
  })

  // 统计错误
  for (const [, agentResult] of result.agentResults) {
    if (!agentResult.success) metrics.errors++
  }

  return { result, metrics }
}

// 使用
const { result, metrics } = await runWithMonitoring(team, goal)

console.log(`
📊 执行报告
══════════════════════════════
LLM 调用: ${metrics.llmCalls}
Token 消耗: ${metrics.totalTokens}
工具调用: ${metrics.toolCalls}
错误数: ${metrics.errors}
耗时: ${Date.now() - metrics.startTime}ms
成功率: ${result.success ? '100%' : '部分失败'}
══════════════════════════════
`)

9.7.2 优雅降级

async function resilientRun(team: Team, goal: string, timeoutMs = 60000) {
  try {
    return await orchestrator.runTeam(team, goal, {
      abortSignal: AbortSignal.timeout(timeoutMs),
    })
  } catch (error) {
    if (error.name === 'AbortError') {
      console.log('⏰ 超时，返回部分结果')

}

9.7.3 资源清理

async function runWithCleanup(team: Team, goal: string) {
  const controller = new AbortController()

  // 确保清理
  const cleanup = () => {
    controller.abort()
    team.removeAllListeners()
    console.log('🧹 资源已清理')
  }

  // 无论如何都要清理
  try {
    return await orchestrator.runTeam(team, goal, {
      abortSignal: controller.signal,
    })
  } finally {
    cleanup()
  }
}

本章小结

  1. 追踪系统（Tracing）：通过 onTrace 回调收集 TraceEvent，可用于调试、成本分析、性能优化。
  1. 流式输出（Streaming）：AsyncGenerator 提供 text/tool_use/tool_result/done/error/loop_detected 事件，实现实时输出和进度显示。
  1. 批量执行：AgentPool 的 runParallel() 和 ToolExecutor 的 executeBatch() 支持并行处理。
  1. AbortSignal：支持取消、超时、组合取消信号，实现可控执行。
  1. 事件系统：TaskQueue 和 Team 提供 task:ready/complete/failed 和 agent:start/complete 等事件。
  1. 可观测性最佳实践：开发/生产环境不同配置、优雅降级、资源清理。
